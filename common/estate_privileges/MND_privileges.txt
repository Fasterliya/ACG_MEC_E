# ============================================================
# 蒙德 MND 专属特权：处理居民委托（挂马拉塔 / 西风骑士团）
# 模板：原版 estate_maratha_development_of_marathi_lands
#       （common/estate_privileges/002_maratha_privileges.txt）
#       —— 同一套 on_granted_province / on_invalid_province / on_revoked_province
#          给低地省份挂 local_development_cost 的常驻省修正
# 省修正体：原版 00_event_modifiers.txt:11751-11756（local_development_cost）
# 地区键：map/region.txt:54 low_countries_region（含 holland_area）
# 登记：common/estates/ACG_MEC_MND_estate.txt 的 estate_maratha -> privileges
# ============================================================

MND_resident_commissions = {
	icon = privilege_development_efficiency
	loyalty = 0.1
	influence = 0.05
	max_absolutism = -5
	conditional_modifier = { trigger = { has_absolutism_reduction_for_estate_privileges = yes } modifier = { max_absolutism = 1 } }
	is_valid = {
		has_dlc = "Domination"
	}
	can_select = {
		tag = MND
		any_owned_province = {
			region = low_countries_region
		}
	}
	on_granted = {
		custom_tooltip = MND_resident_commissions_gt
		tooltip = {
			add_province_modifier = {
				name = MND_resident_commissions_mod
				duration = -1
			}
		}
	}
	on_revoked = {
		custom_tooltip = MND_resident_commissions_rt
	}
	on_granted_province = {
		if = {
			limit = {
				region = low_countries_region
			}
			add_province_modifier = {
				name = MND_resident_commissions_mod
				duration = -1
			}
		}
	}
	on_invalid_province = {
		if = {
			limit = {
				has_province_modifier = MND_resident_commissions_mod
			}
			remove_province_modifier = MND_resident_commissions_mod
		}
	}
	on_revoked_province = {
		if = {
			limit = {
				has_province_modifier = MND_resident_commissions_mod
			}
			remove_province_modifier = MND_resident_commissions_mod
		}
	}
	mechanics = {
		no_estate_crownland_loss_from_developing
	}
	ai_will_do = {
		factor = 0
	}
}
