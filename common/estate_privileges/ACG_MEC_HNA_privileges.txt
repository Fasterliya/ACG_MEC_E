# ============================================================
# 稗田阿求：著书特权（5 本，挂贵族阶层 estate_nobles）
# ------------------------------------------------------------
# 【v3 流程：特权就是按钮】
#   玩家在 **贵族阶层界面** 自己授予「著述《XX》」特权
#     → on_granted 立刻排程一个延时事件：country_event = { id = ... days = N }
#     → 到期事件弹出，玩家三选一拿奖励
#     → 事件的 immediate 里 remove_estate_privilege 收掉特权
#   不再需要任何著书决议，也不再需要年度脉冲计时。
#
# 【特权自带 Debuff】按你最初的要求，Debuff 写在 penalties 里（每本书不同）。
#   上一版临时拆出去的 ACG_MEC_HNA_wdb_1..5 修正已删除。
#
# 【前置条件搬进 can_select】
#   原来写书决议的 allow 里用的是 HNA_can_write_N_trigger（含政体开关 / Mana /
#   Magic_exp / 成长点 / 已成书数）。决议删掉后这些 trigger 会变孤儿，
#   所以全部搬进 can_select —— 这样"能不能写这本书"仍由它们把关，
#   "政体解锁"这一层门槛也保住了。
#
# 【互斥】同样放在 can_select（不用 is_valid）——
#   is_valid 不满足会导致已授予的特权被游戏自动撤销（on_invalid），拿它做互斥会互删。
#
# 【写法依据｜全部实测过】
#   · 特权可写 on_granted / on_revoked / on_invalid：
#       原版 common/estate_privileges/02_noble_privileges.txt:1463-1471
#   · can_select = { } 控制「能否被授予」：
#       原版 common/estate_privileges/02_noble_privileges.txt:1460, 1491
#   · 特权效果块内可写 country_event（国家作用域）：
#       ACG1 common/estate_privileges/01_church_privileges.txt:1027
#
# 【登记】EU4 无特权自动登记机制（实测原版 426 个特权定义全部被
#   common/estates/*.txt 的 privileges 列表引用，孤儿 = 0）。
#   本文件这 5 个 key 已登记在本 mod 的 common/estates/02_nobility.txt
#   （ACG1 该文件的完整覆盖副本 + 5 行）。
# ============================================================

# ---------- 1《求闻史纪》：3 年（1095 天） ----------
ACG_MEC_HNA_writing_1 = {
	icon = privilege_aged_paper
	max_absolutism = -5
	influence = 0.05
	loyalty = 0.05
	is_valid = {
		tag = HNA
	}
	can_select = {
		HNA_can_write_1_trigger = yes
		NOT = { has_estate_privilege = ACG_MEC_HNA_writing_2 }
		NOT = { has_estate_privilege = ACG_MEC_HNA_writing_3 }
		NOT = { has_estate_privilege = ACG_MEC_HNA_writing_4 }
		NOT = { has_estate_privilege = ACG_MEC_HNA_writing_5 }
	}
	on_granted = {
		country_event = { id = ACG_MEC_HNA_events.10 days = 1095 }
	}
	on_revoked = {
	}
	on_invalid = {
	}
	penalties = {
		development_cost = 0.15
		build_cost = 0.15
		technology_cost = 0.05
	}
	benefits = {
	}
	ai_will_do = {
		factor = 0
	}
}
