# 【关键限制｜必须先读】
#   EU4 无法把变量的数值代入 define_general / create_general 的 pip 字段。
#   证据：
#     - 原版只提供 trigger_value:<name> 这种取值语法，且只用于 set_variable /

#
# 【check_variable 语义】check_variable = { which = X value = N } 表示 X >= N

# ============================================================

# ------------------------------------------------------------
# 发放成长点（$amount$ 为成长点数）
# ------------------------------------------------------------
HNA_akyuu_grant_growth_effect = {
	change_variable = { which = HNA_akyuu_growth_points value = $amount$ }
	change_variable = { which = HNA_akyuu_exp value = $amount$ }
	HNA_akyuu_check_tier_effect = yes
}


# ------------------------------------------------------------
# 战斗成长（由 on_battle_won_country 调用）
# 防刷：仅陆战获胜、仅阿求在役、90 天冷却
# ------------------------------------------------------------
HNA_akyuu_battle_growth_effect = {
	# 冷却到期则清 flag（had_country_flag 写法来源：工作区 decisions/ACG_MEC_CHN_decisions.txt:44）
	if = {
		limit = {
			has_country_flag = HNA_akyuu_battle_cd
			had_country_flag = { flag = HNA_akyuu_battle_cd days = 90 }
		}
		clr_country_flag = HNA_akyuu_battle_cd
	}
	if = {
		limit = {
			NOT = { has_country_flag = HNA_akyuu_battle_cd }
			NOT = { has_country_flag = HNA_akyuu_peak_flag }
		}
		set_country_flag = HNA_akyuu_battle_cd
		HNA_akyuu_grant_growth_effect = { amount = 1 }
	}
}


# ------------------------------------------------------------
# 著书计时（由 on_yearly_pulse 调用，1 年 +1）
# 说明：原版 on_action 无 on_monthly_pulse，故用年度脉冲；写成「年」为单位。
# ------------------------------------------------------------
HNA_akyuu_writing_tick_effect = {
	if = {
		limit = { has_country_flag = HNA_writing_1_active }
		change_variable = { which = HNA_writing_1_years value = 1 }
	}
	if = {
		limit = { has_country_flag = HNA_writing_2_active }
		change_variable = { which = HNA_writing_2_years value = 1 }
	}
	if = {
		limit = { has_country_flag = HNA_writing_3_active }
		change_variable = { which = HNA_writing_3_years value = 1 }
	}
	if = {
		limit = { has_country_flag = HNA_writing_4_active }
		change_variable = { which = HNA_writing_4_years value = 1 }
	}
	if = {
		limit = { has_country_flag = HNA_writing_5_active }
		change_variable = { which = HNA_writing_5_years value = 1 }
	}

	# ---- 结算：到点则移除特权 + 触发完成事件 ----
	if = {
		limit = {
			has_country_flag = HNA_writing_1_active
			check_variable = { which = HNA_writing_1_years value = 3 }
		}
		remove_estate_privilege = ACG_MEC_HNA_writing
		clr_country_flag = HNA_writing_1_active
		clr_country_flag = HNA_writing_any
		set_variable = { which = HNA_writing_1_years value = 0 }
		country_event = { id = ACG_MEC_HNA_events.10 days = 1 }
	}
	if = {
		limit = {
			has_country_flag = HNA_writing_2_active
			check_variable = { which = HNA_writing_2_years value = 5 }
		}
		remove_estate_privilege = ACG_MEC_HNA_writing

	}

			has_country_flag = HNA_writing_3_active

		}
		remove_estate_privilege = ACG_MEC_HNA_writing

	}
	if = {
		limit = {
			has_country_flag = HNA_writing_4_active
			check_variable = { which = HNA_writing_4_years value = 4 }
		}
		remove_estate_privilege = ACG_MEC_HNA_writing
