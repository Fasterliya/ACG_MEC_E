### REFRESH TIER ###
SKR_refresh_card_tier_effect = {
	remove_country_modifier = skr_card_tier_0
	remove_country_modifier = skr_card_tier_1
	remove_country_modifier = skr_card_tier_2
	remove_country_modifier = skr_card_tier_3
	remove_country_modifier = skr_card_tier_4
	remove_country_modifier = skr_card_tier_5
	SKR_count_all_cards_effect = yes
	if = {
		limit = { check_variable = { which = skr_total_card_count value = 20 } }
		add_country_modifier = { name = skr_card_tier_5 duration = -1 }
	}
	else_if = {
		limit = { check_variable = { which = skr_total_card_count value = 16 } }
		add_country_modifier = { name = skr_card_tier_4 duration = -1 }
	}
	else_if = {
		limit = { check_variable = { which = skr_total_card_count value = 12 } }
		add_country_modifier = { name = skr_card_tier_3 duration = -1 }
	}
	else_if = {
		limit = { check_variable = { which = skr_total_card_count value = 8 } }
		add_country_modifier = { name = skr_card_tier_2 duration = -1 }
	}
	else_if = {
		limit = { check_variable = { which = skr_total_card_count value = 4 } }
		add_country_modifier = { name = skr_card_tier_1 duration = -1 }
	}
	else = {
		add_country_modifier = { name = skr_card_tier_0 duration = -1 }
	}
}
