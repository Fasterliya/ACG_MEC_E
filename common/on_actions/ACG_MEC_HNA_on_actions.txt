# ---- 战斗成长：阿求亲自带队打赢陆战才计成长 ----
on_battle_won_country = {
	if = {
		limit = {
			tag = HNA
			has_reform = ACG_MEC_HNA_chronicle_reform
			has_leader = "Hieda no Akyuu"
		}
		HNA_akyuu_battle_growth_effect = yes
	}
}

# ---- 兜底：重建预约没人处理时（弹窗一直没点）下个月自动补建 ----
# 为什么需要：整军 / 升级路径会先把旧将撤下，再靠弹窗选项在下一个执行时机重建；
# 若玩家始终不点那个弹窗，阿求就会一直缺席。这里保证最多一个月内必被补回。
on_monthly_pulse = {
	if = {
		limit = {
			tag = HNA
			has_country_flag = HNA_akyuu_rebuild_pending
		}
		if = {
			limit = { NOT = { has_leader = "Hieda no Akyuu" } }
			HNA_akyuu_define_general_effect = yes
		}
		else = {
			clr_country_flag = HNA_akyuu_rebuild_pending
		}
	}
}
