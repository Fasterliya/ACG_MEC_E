# ============================================================
# 稗田阿求：事件
# namespace = ACG_MEC_HNA_events
#   .1      突破（16 级每级一次，本事件可重复触发）
#   .10-.14 五本着书完成（玩家三选一）
#   .30     四维圆满 100/100/100/100
#   .40     阿求出任将领
#   .50     稗田家史笔政体成立
#
# v3 变更（著书完成事件）：
#   · 触发来源 = 著书特权的 on_granted 里排程：
#       country_event = { id = ACG_MEC_HNA_events.1N days = <该书年数> }
#     见 common/estate_privileges/ACG_MEC_HNA_privileges.txt
#   · 每个完成事件都加 trigger = { has_estate_privilege = ... } ——
#     玩家若中途撤销特权，到期时 trigger 不满足，事件不会弹出（防止白拿奖励）。
#   · immediate 里 remove_estate_privilege 收掉特权 + set_country_flag = HNA_book_N_done
#     （成书标记移到 immediate，确保无论选哪个选项都会被置上）。
#
# 所有事件均在 decode_localisation/ACG_MEC_HNA_l_english.yml 补齐文本
#
# v5 变更（与任务树永久化配套）：
#   任务树已把 reward_adm_1 / reward_build_1 / reward_army_1 / reward_army_2 /
#   reward_army_3 / reward_magic_tech / reward_dev_1 全部发成永久（duration = -1）。
#   若事件仍以限时同发**同名**修正，存在把刚拿到的永久 buff 刷回限时的风险
#   （同名事件修正重发属「刷新时长」还是「并存叠加」，Wiki 修正修改.html 事件修正一节
#     与原版文件中均无说明 —— 无定论）。故下列发放一并改永久：
#     .10 选项c reward_army_1、.11 选项a reward_build_1、.11 选项c / .12 选项b reward_army_2、
#     .12 选项c / .13 选项c reward_adm_1、.13 选项a reward_army_3、.14 魔法向 reward_magic_tech
#     （上一轮已改：.10 选项a 与 .40 的 reward_dev_1）
#   保持限时不动的是任务树不发放的两个修正：
#     .11 选项b reward_magic_mana（3650）、.12 选项a reward_magic_exp（5475）
#
# v6 变更（著书可重复刷取）：
#   .10 / .11 / .12 / .13 / .14 五个成书事件**移除 fire_only_once = yes**。
#   原因：fire_only_once 是「重复著书」的硬阻塞（Wiki 事件修改.html：「你可以添加如下语句
#   使事件只触发一次」）。第二次授予特权后，排程的事件被静默丢弃，immediate 里的
#   remove_estate_privilege 不执行 → 特权永久挂在贵族身上（连带其 penalties：见
#   common/estate_privileges/ACG_MEC_HNA_privileges.txt 各书的 penalties 块）；
#   而每本书的 can_select 又写着 NOT = { has_estate_privilege = 其它 4 本 }，
#   于是整个著书系统死锁，只能手动撤销特权才能恢复。
#   改动后流程：授予特权 → N 天后成书事件 → immediate 收掉特权 + 置 HNA_book_N_done →
#   玩家三选一拿奖励 → 特权回到可授予状态，可再次著书（互斥保留，故为「顺序重复刷」）。
#   仍保持一次性的：.30（四维圆满）、.40（召阿求出任将领）、.50（稗田家史笔政体成立）。
#   ⚠ 已知风险（无定论）：重复刷会反复发放**同名永久修正**（如 .10 选项a 的
#     ACG_MEC_HNA_reward_dev_1，duration = -1）。同名修正重发是「刷新时长」还是
#     「并列叠加」在 Wiki 与原版文件中均无说明 —— 若为叠加语义，反复刷同一选项会让
#     对应数值无限累积，需实测确认（这正是你选择「奖励保持原样、无限刷」时接受的取舍）。
# ============================================================

# ------------------------------------------------------------
country_event = {
	id = ACG_MEC_HNA_events.10
	title = "ACG_MEC_HNA_events.10.t"
	desc = "ACG_MEC_HNA_events.10.d"
	picture = COMET_SIGHTED_eventPicture
	is_triggered_only = yes
	# v6：移除 fire_only_once，允许重复著书刷取奖励（理由见文件头 v6 段）
	trigger = {
		has_estate_privilege = ACG_MEC_HNA_writing_1
	}
	immediate = {
		remove_estate_privilege = ACG_MEC_HNA_writing_1
		set_country_flag = HNA_book_1_done
	}
	option = {   #发展向
		name = "ACG_MEC_HNA_events.10.a"
		# v5 变更：duration 3650 → -1（永久）。
		# 原因：任务 2_1 / 2_2 与事件 .40 已把同一个 dev_1 发成永久（duration = -1），
		# 若此处仍发 3650 天的同名修正，存在把永久 buff 刷回限时的风险
		# （同名事件修正重发的语义「刷新时长」还是「并存叠加」在 Wiki
		#   修正修改.html 事件修正一节与原版文件中均无说明 —— 无定论）。
		# 三处统一为永久后，无论游戏取哪种语义都不会出问题。
		add_country_modifier = { name = ACG_MEC_HNA_reward_dev_1 duration = -1 }
		HNA_akyuu_grant_growth_effect = { amount = 4 }
		ai_chance = { factor = 1 }
	}
	option = {   #内政向
		name = "ACG_MEC_HNA_events.10.b"
		add_adm_power = 100
		add_stability = 1
		HNA_akyuu_grant_growth_effect = { amount = 4 }
		ai_chance = { factor = 1 }
	}
	option = {   #军事向
		name = "ACG_MEC_HNA_events.10.c"
		add_country_modifier = { name = ACG_MEC_HNA_reward_army_1 duration = -1 }
		add_mil_power = 100
		HNA_akyuu_grant_growth_effect = { amount = 4 }
		ai_chance = { factor = 1 }
	}
}
