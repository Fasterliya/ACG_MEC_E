# ============================================================
# 稗田阿求（HNA）任务树
# ------------------------------------------------------------
# 【列 / 行 语义｜已核对】
#   本地 Wiki 任务树修改.html（EVERGREEN）：
#     slot     = 列（"此任务树所在的槽位，是几就是从左到右第几列，范围为 1-5"）
#     position = 行（"从上到下数本任务的位置，最顶端对应的数值为 1"）
#
# 【箭头硬规则｜本 mod 约定：零越界】
#   任务名记作 X_Y（X = 列 = slot，Y = 行 = position）。EU4 只支持两种连接：
#
#     关系 A｜同列（X 相同）：Δ列 = 0，Δ行 ≥ 1
#        同列下方的任意任务（Y+N，N≥1）都能挂到上方的任务（Y）上。
#
#     关系 B｜跨列（X 不同）：Δ列 ≠ 0，且 Δ行 必须恰好 = 1
#        只有"上一行（Y−1）"的任务能挂到（X, Y）上；列距不限（Δ列 可 ±1、±2、±3…）。
#
#   实测依据（原版 1.37.5，3709 个任务、约 2800 条连接）：
#     (0,+1)(0,+2)(0,+3)(0,+4)  1656 次  59.1%   ← 关系 A
#     (±1,+1)                    970 次  34.6%   ← 关系 B
#     (±2,+3,+1)                  88 次   3.1%   ← 关系 B（跳列，合法）
#     而 Δ行 = 0 有 49 次（1.7%）、Δ行 < 0 有 3 次 —— 那是原版自己的越界写法
#     （例：KoK_Byzantine_Missions.txt 的 BYZ_warehouse_of_the_east
#        [slot4,pos12] 同时依赖 [slot3,pos12] 与 [slot5,pos12]），
#     本 mod 明确不采用，全部按 0 越界排布。
#
# 【空间规划】不必把每个 (X, Y) 都填满 —— 可以留白做视觉规划。
#
#   本树采用「阶梯错位」：
#
#     行1   2_1 稗田家的领地（唯一根，无前置）
#             │ 竖
#     行2   2_2 五处聚落 ──斜──► 3_1 城防与治安
#             │ 竖                  │ 竖
#     行3   2_3 书库与学舍          3_2 安科纳的商路 ──斜──► 4_1 《求闻史纪》
#             │ 竖                  │ 竖                     │ 竖
#     行4   2_4 印刷工坊            3_3 亚平宁的立足          4_2 《幻想乡缘起》
#                                   │ 竖                     │ 竖
#     行5                            3_4 罗马之路 ──斜──►    4_3 魔法与史笔
#                                                            │ 竖   ▲ 斜（来自 3_4）
#     行6                                                     ▼      │
#                                                           4_4 幻想乡史笔
#
#     留白：(列2, 行5-6)、(列3, 行1, 行6)、(列4, 行1-2)、(列5, 整列)
#     列2 占行1-4，列3 占行2-5，列4 占行3-6 —— 三列各错开一行。
#
#   全部 11 条连接：(Δ列, Δ行) 只有 (0,1) 与 (1,1) 两种 → 零越界。
#
# 【图标】icon = HNA_mission_N → 本 mod interface/ACG_MEC_E_HNA.gfx 里的同名 spriteType
#   本 mod 家族惯例：任务图标 spriteType 不带 GFX_ 前缀
#   （ACG1 interface/IMS_adviser.gfx:275-278 name = "LLF_icon_1"，
#     missions/IMS_misssions.txt:1513 icon = LLF_icon_1；
#     原版 interface/countrymissionsview.gfx:526 name = "mission_noble_council"）
#   尺寸：59 x 63
#
# 【本地化】键名必须是 <任务名>_title 与 <任务名>_desc
#   （依据：原版 decisions↔loc 命名规范同族；工作区 HKH_l_english.yml、ALC_l_english.yml 均为该格式）
# ============================================================

# 第 4 列（slot = 4）：著书 / 魔法     占行 3-6

HNA_missions_4 = {
	slot = 4
	generic = no
	ai = yes
	potential = {
		tag = HNA
		NOT = { map_setup = map_setup_random }
	}
	has_country_shield = yes

	# ---- 行3：《求闻史纪》（关系 B：斜挂在列3行2下；需先解锁政体）----
	HNA_missions_4_1 = {
		icon = HNA_mission_9
		position = 3
		required_missions = { HNA_missions_3_1 }
		trigger = {
			has_reform = ACG_MEC_HNA_chronicle_reform
			has_country_flag = HNA_book_1_done
		}
		effect = {
			add_country_modifier = { name = ACG_MEC_HNA_reward_adm_1 duration = 3650 }
			add_splendor = 100
			HNA_akyuu_grant_growth_effect = { amount = 6 }
		}
	}

	# ---- 行4：《幻想乡缘起》----
	HNA_missions_4_2 = {
		icon = HNA_mission_10
		position = 4
		required_missions = { HNA_missions_4_1 }
		trigger = {
			has_country_flag = HNA_book_2_done
		}
		effect = {
			add_country_modifier = { name = ACG_MEC_HNA_reward_build_1 duration = 3650 }
			HNA_akyuu_grant_growth_effect = { amount = 5 }
		}
	}

	# ---- 行5：魔法与史笔 ----
	HNA_missions_4_3 = {
		icon = HNA_mission_11
		position = 5
		required_missions = { HNA_missions_4_2 }
		trigger = {
			check_variable = { which = Mana value = 25 }
			check_variable = { which = Magic_exp value = 10 }
		}
		effect = {
			change_variable = { which = Mana value = 20 }
			change_variable = { which = Magic_exp value = 15 }
			add_country_modifier = { name = ACG_MEC_HNA_reward_magic_tech duration = 5475 }
			HNA_akyuu_grant_growth_effect = { amount = 5 }
		}
	}

	# ---- 行6：幻想乡史笔（终点；双前置，两条都合法）
	#       4_3  同列 Δ列=0、Δ行=1  → 关系 A
	#       3_4  跨列 Δ列=1、Δ行=1  → 关系 B
	#      —— 这样既完成收束，又让"军事线"真正汇入"著书线"。
	HNA_missions_4_4 = {
		icon = HNA_mission_12
		position = 6
		required_missions = {
			HNA_missions_4_3
			HNA_missions_3_4
		}
		trigger = {
			check_variable = { which = HNA_akyuu_growth_points value = 160 }
		}
		effect = {
			add_country_modifier = { name = ACG_MEC_HNA_chronicle_permanent duration = -1 }
			add_prestige = 50
			HNA_akyuu_grant_growth_effect = { amount = 5 }

			# ---------- v4 新增 ①：全科技 +1 ----------
			# add_adm_tech / add_dip_tech / add_mil_tech 是原版 effect：
			#   原版 events/flavorPOL.txt:2143（add_adm_tech = 1）
			#   原版 events/FlavorRUS.txt:2285 / 2299 / 2313（adm / dip / mil 各一）
			#   原版 missions/DOM_Chinese_Missions.txt:2556（add_mil_tech = 1）
			#   Wiki 指令.html §科技：「提升行政科技等级」「该指令无法降低科技」
			# 用 hidden_effect 包住 → 不让游戏再自动生成三行同样的 tooltip，
			# 统一交给下面那条 custom_tooltip 叙述。
			# （custom_tooltip 与 hidden_effect 都是原版 missions 的高频写法：
			#   custom_tooltip 4015 处、hidden_effect 968 处。想改回自动 tooltip
			#   就把 hidden_effect 这层去掉即可。）
			hidden_effect = {
				add_adm_tech = 1
				add_dip_tech = 1
				add_mil_tech = 1
			}

			# ---------- v4 新增 ②：解锁阿求「再招募」机制 ----------
			# 置此 flag 后：
			#   · 旧决议 ACG_MEC_HNA_summon_akyuu 不再出现（其 potential 排除了已解锁状态）
			#   · 新决议 ACG_MEC_HNA_recall_akyuu 出现 —— 不受战争状态限制、冷却 30 天
			#   · 阿求累积的成长点 / 突破等级是**国家变量**，所以再召回来的她
			#     会以同样的四维等级重建（HNA_akyuu_rebuild_general_effect 读 HNA_akyuu_tier）
			set_country_flag = HNA_akyuu_recall_unlocked
			custom_tooltip = ACG_MEC_HNA_final_mission_tt
		}
	}
}
