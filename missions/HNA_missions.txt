# ============================================================
# 稗田阿求（HNA）任务树
# ------------------------------------------------------------
# 【列 / 行 语义｜已核对】
#   本地 Wiki 任务树修改.html（EVERGREEN）：
#     slot     = 列（"此任务树所在的槽位，是几就是从左到右第几列，范围为 1-5"）
#     position = 行（"从上到下数本任务的位置，最顶端对应的数值为 1"）
#
# 【箭头硬规则｜本 mod 约定：零越界】
#   任务名记作 X_Y（X = 列 = slot，Y = 行 = position）。EU4 只支持两种连接：
#
#     关系 A｜同列（X 相同）：Δ列 = 0，Δ行 ≥ 1
#        同列下方的任意任务（Y+N，N≥1）都能挂到上方的任务（Y）上。
#
#     关系 B｜跨列（X 不同）：Δ列 ≠ 0，且 Δ行 必须恰好 = 1
#        只有"上一行（Y−1）"的任务能挂到（X, Y）上；列距不限（Δ列 可 ±1、±2、±3…）。
#
#   实测依据（原版 1.37.5，3709 个任务、约 2800 条连接）：
#     (0,+1)(0,+2)(0,+3)(0,+4)  1656 次  59.1%   ← 关系 A
#     (±1,+1)                    970 次  34.6%   ← 关系 B
#     (±2,+3,+1)                  88 次   3.1%   ← 关系 B（跳列，合法）
#     而 Δ行 = 0 有 49 次（1.7%）、Δ行 < 0 有 3 次 —— 那是原版自己的越界写法
#     （例：KoK_Byzantine_Missions.txt 的 BYZ_warehouse_of_the_east
#        [slot4,pos12] 同时依赖 [slot3,pos12] 与 [slot5,pos12]），
#     本 mod 明确不采用，全部按 0 越界排布。
#
# 【空间规划】不必把每个 (X, Y) 都填满 —— 可以留白做视觉规划。
#
#   本树采用「阶梯错位」：
#
#     行1   2_1 稗田家的领地（唯一根，无前置）
#             │ 竖
#     行2   2_2 五处聚落 ──斜──► 3_1 城防与治安
#             │ 竖                  │ 竖
#     行3   2_3 书库与学舍          3_2 安科纳的商路 ──斜──► 4_1 《求闻史纪》
#             │ 竖                  │ 竖                     │ 竖
#     行4   2_4 印刷工坊            3_3 亚平宁的立足          4_2 《幻想乡缘起》
#                                   │ 竖                     │ 竖
#     行5                            3_4 罗马之路 ──斜──►    4_3 魔法与史笔
#                                                            │ 竖   ▲ 斜（来自 3_4）
#     行6                                                     ▼      │
#                                                           4_4 幻想乡史笔
#
#     留白：(列2, 行5-6)、(列3, 行1, 行6)、(列4, 行1-2)、(列5, 整列)
#     列2 占行1-4，列3 占行2-5，列4 占行3-6 —— 三列各错开一行。
#
#   全部 11 条连接：(Δ列, Δ行) 只有 (0,1) 与 (1,1) 两种 → 零越界。
#
# 【图标｜v7 变更】全部改用原版通用图标池的 spriteType 名（不再依赖本 mod 自建图）
#   原因：原先 12 个 icon = HNA_mission_N 指向 gfx/interface/missions/HNA/*.dds，
#         该目录/文件不存在，导致每次启动刷 56,712 行 spritetype 报错
#         （error.log：[spritetype.cpp:339] Error initialising texture … 每个 3074 次；
#           [spritetype.cpp:522] Should not be lazy loaded … 每个 1652 次）。
#   证据（VANILLA_VERIFIED，1.37.5）：
#     原版 interface/countrymissionsview.gfx 定义任务图标 spriteType
#       例：:279 name = "mission_early_game_buildings"
#           :280 texturefile = "gfx//interface//missions//mission_early_game_buildings.dds"
#     12 个图标的定义与 dds 均已逐一核对存在。
#   尺寸：59 x 63（原版池统一规格）
#
# 【本地化】键名必须是 <任务名>_title 与 <任务名>_desc
#   （依据：原版 decisions↔loc 命名规范同族；工作区 HKH_l_english.yml、ALC_l_english.yml 均为该格式）
# ============================================================
#
# 【v5 变更｜本轮】
#   1) 任务树给的修正**全部改为永久**（duration = -1）：
#        2_3 reward_adm_1、2_4 reward_build_1、3_1 reward_army_1、3_2 reward_dev_2、
#        3_3 reward_army_2、3_4 reward_army_3、4_1 reward_adm_1、4_2 reward_build_1、
#        4_3 reward_magic_tech
#      （2_1 / 2_2 的 reward_dev_1 与 4_4 的 chronicle_permanent 本来就是 duration = -1）
#      理由：这是长线养成树，限时奖励 10~15 年就失效，与「永久成长」的意图不符。
#   2) 卡后期科技的建筑条件改掉（科技 17 = 1622 年，太靠后）：
#        · 2_3「书库与学舍」：has_building = university（行政 17）→ temple（行政 4）
#        · 3_2「安科纳的商路」：has_building = trade_depot（外交 17）→ marketplace（外交 4）
#      科技门槛证据见各自任务上方注释。2_4（工坊 行政 6 + 集市 外交 4）与
#      3_1（法院 行政 8 + 兵营 军事 6）属前期~中期，按你的要求保留不动。
#   3) 同名修正撞车防护：事件里与任务永久版同名的限时发放也一并改永久，
#      见 events/ACG_MEC_HNA_events.txt 头部的 v5 说明。
#
# 【v6 变更｜本轮】
#   1) 2_1「稗田家的领地」开局赠送 add_treasury = 2000（原版同额实例
#      missions/DOM_Britain_Missions.txt:4358）。
#   2) 军事线三颗（3_1 → 3_3 → 3_4）各追加两份永久修正，共 3 次：
#        · 骰子：ACG_MEC_HNA_dice_1 / _2 / _3，每个都是
#            global_attacker_dice_roll_bonus = 1
#            global_defender_dice_roll_bonus = 1
#          → 三次累计 **全局进攻骰子 +3、全局防御骰子 +3**。
#          键名证据（1.34+ 加法型修正）：Wiki 修正列表「全局进攻方骰子增加」/「全局防御方骰子增加」；
#          原版实例 common/event_modifiers/01_mission_modifiers.txt:8805 / :9803。
#          ⚠ 为什么用 3 个独立键而不是同一个键发 3 次：同名修正重发是「刷新时长」还是
#            「并列叠加」，Wiki 修正修改.html 事件修正一节与原版文件中均无说明（无定论），
#            共用一键有可能只生效 +1。独立键在两种语义下都必然叠到 +3。
#        · 军事包：ACG_MEC_HNA_mil_1 / _2 / _3（见 common/event_modifiers/ACG_MEC_HNA_modifiers.txt）
#          均为**新建独立键**，不改动既有 reward_army_1/2/3，
#          因此事件（.10c / .11c / .12b / .13a）不会连带变强。
#   3) 每月君主点数分 2 次给：2_2「五处聚落」给 ACG_MEC_HNA_power_1，
#      4_4「幻想乡史笔」给 ACG_MEC_HNA_power_2，每个键都是
#        country_admin_power = 1 / country_diplomatic_power = 1 / country_military_power = 1
#      → 两次合计 **每月行政 +2、外交 +2、军事 +2**。
#      键名证据（加法型修正）：Wiki 修正列表「每月额外获得行政/外交/军事君主力量点数」
#      （1.26 / 1.30）；原版实例 common/event_modifiers/01_mission_modifiers.txt:9867-9868、
#      common/event_modifiers/00_event_modifiers.txt:13160-13162、
#      common/static_modifiers/00_static_modifiers.txt:1910-1918。
#      同样拆成 2 个独立键，理由与骰子一致（同名重发语义无定论）。

HNA_missions_2 = {
	slot = 2
	generic = no
	ai = yes
	potential = {
		tag = HNA
		NOT = { map_setup = map_setup_random }
	}
	has_country_shield = yes

	# ---- 行1：稗田家的领地（全树唯一根）----
	#   v5 变更：trigger 由「首都发展度 20」改为 always = yes —— 开局即可完成，
	#     从而在游戏一开始就拿到「意大利统一战争」宣战理由（见下方 effect）。
	#     always = yes 作为任务 trigger 的原版实例（VANILLA_VERIFIED）：
	#       原版 missions/AFR_West_African_Missions.txt:473-474
	HNA_missions_2_1 = {
		icon = mission_unite_home_region
		position = 1
		required_missions = { }
		trigger = {
			always = yes
		}
		effect = {
			# v6 变更：开局赠款 2000 杜卡特。
			#   add_treasury 为原版效果；同额原版实例 missions/DOM_Britain_Missions.txt:4358。
			add_treasury = 2000
			capital_scope = {
				add_base_tax = 1
			}
			# v5 变更：duration 3650 → -1（永久 -15% 发展成本）
			add_country_modifier = { name = ACG_MEC_HNA_reward_dev_1 duration = -1 }
			HNA_akyuu_grant_growth_effect = { amount = 3 }

			# ---------- 解锁「意大利统一战争」宣战理由 ----------
			# 与 YYK 起始任务同款写法（missions/YYK_missions.txt:19-20）：置 flag + 自定义 tooltip。
			# 该 CB 本体 common/cb_types/YYK_cb_types.txt:4-12 的 prerequisites 只判
			#     has_country_flag = yyk_italian_cb_unlocked_flag
			#     （OR 的另一支 mission_completed = YYK_missions_1_1 与本行无关）
			# 不判 tag，所以 HNA 置上同一个 flag 即可使用此 CB；
			# 另一条 prerequisites（any_known_country 持有 italy_region 省份）由时局自然满足。
			# CB 附带效果：-75% 侵略扩张、-50% 战争分数花费、防御方无法召唤盟友
			#   （挡援军写在 common/diplomatic_actions/00_diplomatic_actions.txt:1020-1033）
			# tooltip 复用 YYK 已编码好的文本（decode_localisation/YYK_l_english.yml:93），
			# 正文 tag 中立，故本任务无需新增本地化 key。
			set_country_flag = yyk_italian_cb_unlocked_flag
			custom_tooltip = yyk_italian_cb_unlocked_tt
		}
	}

	# ---- 行2：五处聚落（关系 A：同列 Δ行=1）----
	HNA_missions_2_2 = {
		icon = mission_early_game_buildings
		position = 2
		required_missions = { HNA_missions_2_1 }
		trigger = {
			num_of_cities = 5
			any_owned_province = {
				development = 15
				has_owner_culture = yes
			}
		}
		effect = {
			# v5 变更：duration 3650 → -1（永久 -15% 发展成本）
			add_country_modifier = { name = ACG_MEC_HNA_reward_dev_1 duration = -1 }
			# v6：每月君主点数第 1 次 —— 行政 / 外交 / 军事 各 +1（ACG_MEC_HNA_power_1）
			add_country_modifier = { name = ACG_MEC_HNA_power_1 duration = -1 }
			add_adm_power = 50
			HNA_akyuu_grant_growth_effect = { amount = 4 }
		}
	}

	# ---- 行3：书库与学舍 ----
	HNA_missions_2_3 = {
		icon = mission_early_modern_university
		position = 3
		required_missions = { HNA_missions_2_2 }
		# v5 变更：原条件 has_building = university（大学）改为 temple（庙宇）。
		#   大学由 **行政科技 17 / 1622 年** 解锁（VANILLA_VERIFIED：
		#     原版 common/technologies/adm.txt:248-259，块注释「#University」「# Tech 17」），
		#     在这棵前期任务树里等于卡到 17 世纪；
		#   庙宇由 **行政科技 4 / 1453 年** 解锁（原版 common/technologies/adm.txt:70-78），
		#     开局不久即可达成，保留「要盖东西」的手感。
		#   （任务描述《书库与学舍》的文案未动，仍在 decode_localisation。）
		trigger = {
			capital_scope = {
				has_building = temple
			}
		}
		effect = {
			add_country_modifier = { name = ACG_MEC_HNA_reward_adm_1 duration = -1 }
			add_splendor = 50
			HNA_akyuu_grant_growth_effect = { amount = 5 }
		}
	}

	# ---- 行4：印刷工坊 ----
	HNA_missions_2_4 = {
		icon = mission_have_manufactories
		position = 4
		required_missions = { HNA_missions_2_3 }
		trigger = {
			any_owned_province = {
				has_building = workshop
			}
			any_owned_province = {
				has_building = marketplace
			}
		}
		effect = {
			add_country_modifier = { name = ACG_MEC_HNA_reward_build_1 duration = -1 }
			add_treasury = 100
			HNA_akyuu_grant_growth_effect = { amount = 4 }
		}
	}
}
