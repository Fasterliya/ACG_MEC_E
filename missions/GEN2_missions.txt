gen2_military = {
	slot = 6
	generic = yes
	ai = no
	potential = {
		always = yes
	}
	has_country_shield = no

	gen2_build_army = {
		icon = mission_assemble_an_army
		position = 1
		provinces_to_highlight = {
			OR = {
				NOT = { owned_by = ROOT }
				NOT = { is_claim = ROOT }
				NOT = { is_permanent_claim = ROOT }
			}
		}
		trigger = {
			army_size_percentage = 1.0
		}
		effect = {
			custom_tooltip = perma_claim_neighbor_area
			hidden_effect = {
				every_owned_province = {
					every_neighbor_province = {
						area = {
							limit = {
								NOT = { owned_by = ROOT }
								NOT = { is_permanent_claim = ROOT }
								NOT = { is_core = ROOT }
							}
							add_permanent_claim = ROOT
						}
					}
				}
			}
			add_country_modifier = {
				name = "gen2_military_readiness"
				duration = -1
			}
			add_splendor = 50
			add_prestige = 10
			add_mil_power = 100
			change_government_reform_progress = 25
		}
	}
	gen2_expand_country = {
		icon = mission_unite_home_region
		position = 2
		required_missions = { gen2_build_army }
		provinces_to_highlight = {
			OR = {
				NOT = { owned_by = ROOT }
				NOT = { is_claim = ROOT }
				NOT = { is_permanent_claim = ROOT }
			}
		}
		trigger = {
			OR = {
				grown_by_development = 50
				army_size_percentage = 0.9
				AND = {
					ai = yes
					total_development = 150
				}
			}
		}
		effect = {
			custom_tooltip = perma_claim_neighbor_area
			hidden_effect = {
				every_owned_province = {
					every_neighbor_province = {
						area = {
							limit = {
								NOT = { owned_by = ROOT }
								NOT = { is_permanent_claim = ROOT }
								NOT = { is_core = ROOT }
							}
							add_permanent_claim = ROOT
						}
					}
				}
			}
			add_splendor = 100
			add_prestige = 15
			add_mil_power = 150
			add_army_tradition = 20
		}
	}
	gen2_assemble_grand_army = {
		icon = mission_build_up_to_force_limit
		position = 3
		required_missions = { gen2_expand_country }
		trigger = {
			OR = {
				army_size = 60
				land_forcelimit = 80
				AND = {
					ai = yes
					army_size_percentage = 0.8
				}
			}
		}
		effect = {
			if = {
				limit = { NOT = { army_tradition = 40 } }
				create_general = { tradition = 40 }
			}
			else_if = {
				limit = { NOT = { army_tradition = 60 } }
				create_general = { tradition = 60 }
			}
			else_if = {
				limit = { NOT = { army_tradition = 80 } }
				create_general = { tradition = 80 }
			}
			else = {
				create_general = { tradition = 100 }
			}
			add_country_modifier = {
				name = "gen2_grand_army"
				duration = -1
			}
			add_splendor = 150
			add_prestige = 20
			add_mil_power = 200
		}
	}
	gen2_military_hegemony = {
		icon = mission_cannons_firing
