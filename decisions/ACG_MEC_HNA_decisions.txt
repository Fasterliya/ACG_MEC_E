# ============================================================
# 稗田阿求：决议
# ------------------------------------------------------------
# 说明：本 mod 现有决议文件均不带 icon 字段，此处保持一致。
# 本地化键名规范：<key>_title + <key>_desc
#   （原版证据：decisions/AbolishTheElectiveMonarchy.txt
#     ↔ localisation/winds_of_change_l_english.yml 的
#       boh_abolish_the_elective_monarchy_title / _desc）
# 效果名来源：set_estate_privilege（原版 85 处）/ remove_estate_privilege（171 处）
#            add_government_reform（工作区 history/countries/*.txt）
#
# v3 变更：5 条「著书」决议已全部删除 ——
#          著书改为玩家直接在 **贵族阶层界面** 授予著书特权
#          （见 common/estate_privileges/ACG_MEC_HNA_privileges.txt）。
#          本文件只保留 3 条与著书无关的决议。

country_decisions = {

	# --------------------------------------------------------
	# 1. 采纳「稗田家史笔」政体 → 解锁阿求成长系统
	# --------------------------------------------------------
	#   v5 变更：改为「开局白送」——
	#     · allow 由 { has_reform = Touhou_Republic_government_reform, prestige = 10 }
	#       改为 always = yes：不再要求先持有 Touhou_Republic 政体，也不再有 10 声望门槛。
	#       always = yes 作为决议 allow 的原版实例（VANILLA_VERIFIED）：
	#         原版 decisions/Governments.txt:226-228（决议 stop_promoting_art）
	#     · effect 移除 add_prestige = -20（原为「采纳政体」的声望代价）→ 真正零代价。
	#   potential 保留 tag = HNA 与 NOT has_reform —— 仍限定 HNA，且只能采纳一次。
	#   （注：政体自身给的长驻 buff 在 common/government_reforms/ACG_MEC_HNA_reforms.txt:24-29，
	#     本次未改动；本决议触发的 ACG_MEC_HNA_events.50 本来就只给 add_splendor = 100。）
	ACG_MEC_HNA_adopt_chronicle_reform = {
		potential = {
			tag = HNA
			NOT = { has_reform = ACG_MEC_HNA_chronicle_reform }
		}
		allow = {
			always = yes
		}
		effect = {
			add_government_reform = ACG_MEC_HNA_chronicle_reform
			set_country_flag = HNA_akyuu_growth_enabled
			country_event = { id = ACG_MEC_HNA_events.50 }
		}
		ai_will_do = {
			factor = 20
		}
	}

	# --------------------------------------------------------
	# 2. 召阿求出任将领（决议生成的稗田阿求将领）
	# --------------------------------------------------------
	ACG_MEC_HNA_summon_akyuu = {
		potential = {
			tag = HNA
			has_reform = ACG_MEC_HNA_chronicle_reform
			NOT = { has_leader = "Hieda no Akyuu" }
			# 完成最终任务（幻想乡史笔）后，本决议让位给「再召稗田阿求」
			NOT = { has_country_flag = HNA_akyuu_recall_unlocked }
		}
		allow = {
			is_at_war = no
		}
		effect = {
			# 初始化成长变量（只做一次）
			if = {
				limit = { NOT = { has_country_flag = HNA_akyuu_init } }
				set_country_flag = HNA_akyuu_init
				set_variable = { which = HNA_akyuu_growth_points value = 0 }
				set_variable = { which = HNA_akyuu_tier value = 0 }
				set_variable = { which = HNA_akyuu_exp value = 0 }
			}
			HNA_akyuu_rebuild_general_effect = yes
			country_event = { id = ACG_MEC_HNA_events.40 }
		}
		ai_will_do = {
			factor = 0
		}
	}

	# --------------------------------------------------------
	# 3. 整军：按当前成长点手动重建阿求（兜底 + 玩家自主刷新）
	# --------------------------------------------------------
	ACG_MEC_HNA_refit_akyuu = {
		potential = {
			tag = HNA
			has_reform = ACG_MEC_HNA_chronicle_reform
			has_leader = "Hieda no Akyuu"
		}
		allow = {
			OR = {
				NOT = { has_country_flag = HNA_akyuu_refit_cd }
				had_country_flag = { flag = HNA_akyuu_refit_cd days = 365 }
			}
		}
		effect = {
			clr_country_flag = HNA_akyuu_refit_cd
			set_country_flag = HNA_akyuu_refit_cd
			HNA_akyuu_rebuild_general_effect = yes
		}
		ai_will_do = {
			factor = 0
		}
	}

	# --------------------------------------------------------
	# 4. 再召稗田阿求（v4 新增；由最终任务「幻想乡史笔」解锁）
	# --------------------------------------------------------
	# 与第 2 条的区别（两者靠 HNA_akyuu_recall_unlocked 互斥，不会同时出现）：
	#   第 2 条：未解锁时用，必须处于和平，无冷却
	#   本  条：解锁后用，**不受战争状态限制**，冷却 30 天
	# 阿求的成长点 / 突破等级存在国家变量里，所以再召回来仍是同一四维等级
	# （HNA_akyuu_rebuild_general_effect 按 HNA_akyuu_tier 查表重建）。
	ACG_MEC_HNA_recall_akyuu = {
		potential = {
			tag = HNA
			has_reform = ACG_MEC_HNA_chronicle_reform
			has_country_flag = HNA_akyuu_recall_unlocked
			NOT = { has_leader = "Hieda no Akyuu" }
		}
		allow = {
			OR = {
				NOT = { has_country_flag = HNA_akyuu_recall_cd }
				had_country_flag = { flag = HNA_akyuu_recall_cd days = 30 }
			}
		}
		effect = {
			# 与第 2 条同样的初始化保护（正常流程下变量已存在，此处仅兜底）
			if = {
				limit = { NOT = { has_country_flag = HNA_akyuu_init } }
				set_country_flag = HNA_akyuu_init
				set_variable = { which = HNA_akyuu_growth_points value = 0 }
				set_variable = { which = HNA_akyuu_tier value = 0 }
				set_variable = { which = HNA_akyuu_exp value = 0 }
			}
			clr_country_flag = HNA_akyuu_recall_cd
			set_country_flag = HNA_akyuu_recall_cd
			HNA_akyuu_rebuild_general_effect = yes
		}
		ai_will_do = {
			factor = 0
		}
	}
}
