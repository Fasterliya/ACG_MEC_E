		potential = {
			tag = HNA
			NOT = { has_reform = ACG_MEC_HNA_chronicle_reform }
		}

			has_reform = Touhou_Republic_government_reform

			NOT = { has_leader = "Hieda no Akyuu" }
		}
		allow = {
			is_at_war = no
		}
		effect = {
			# 初始化成长变量（只做一次）
			if = {
				limit = { NOT = { has_country_flag = HNA_akyuu_init } }
				set_country_flag = HNA_akyuu_init
				set_variable = { which = HNA_akyuu_growth_points value = 0 }
				set_variable = { which = HNA_akyuu_tier value = 0 }
				set_variable = { which = HNA_akyuu_exp value = 0 }
			}
			HNA_akyuu_rebuild_general_effect = yes
			country_event = { id = ACG_MEC_HNA_events.40 }
		}
		ai_will_do = {
			factor = 0
		}
	}

	ACG_MEC_HNA_refit_akyuu = {
		potential = {
			tag = HNA
			has_reform = ACG_MEC_HNA_chronicle_reform
			has_leader = "Hieda no Akyuu"
		}
		allow = {
			OR = {
				NOT = { has_country_flag = HNA_akyuu_refit_cd }
				had_country_flag = { flag = HNA_akyuu_refit_cd days = 365 }
			}
		}
		effect = {
			clr_country_flag = HNA_akyuu_refit_cd
			set_country_flag = HNA_akyuu_refit_cd
			HNA_akyuu_rebuild_general_effect = yes
		}
		ai_will_do = {
			factor = 0
		}
	}

	# --------------------------------------------------------
	ACG_MEC_HNA_write_1 = {

			tag = HNA

		}
		allow = {

		}
