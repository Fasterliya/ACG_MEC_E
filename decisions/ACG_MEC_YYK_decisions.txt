country_decisions = {
	ACG_MEC_yyk_culture_calibrate = {
		potential = {
			tag = YYK
			ai = no
		}
		allow = {
			always = yes
		}
		effect = {
			hidden_effect = {
				# Register one not-yet-registered culture present in owned provinces per loop pass
				while = {
					limit = {
						any_owned_province = {
							NOT = { culture = ROOT }
							NOT = { has_province_flag = acg_mec_yyk_known_culture@ROOT }
						}
					}
					random_owned_province = {
						limit = {
							NOT = { culture = ROOT }
							NOT = { has_province_flag = acg_mec_yyk_known_culture@ROOT }
						}
						save_event_target_as = acg_mec_yyk_culture_prov
					}
					# Flag every province of that culture so it is never counted twice
					every_province = {
						limit = { culture = event_target:acg_mec_yyk_culture_prov }
						set_province_flag = acg_mec_yyk_known_culture@ROOT
					}
					# +1 netherworld power per newly detected culture
					add_government_power = {
						mechanic_type = YYK_netherworld_mechanic
						power_type = YYK_netherworld_power
						value = 1
					}
				}
			}
		}
		ai_will_do = {
			factor = 0
		}
	}
}
