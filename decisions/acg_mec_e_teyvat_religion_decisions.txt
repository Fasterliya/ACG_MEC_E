# ACG_MEC_E - Teyvat: switch personal deity
# 2 decisions -> 2 events (options pick the new deity)
# 本地化键名规范：<key>_title + <key>_desc
# 重择不消耗正统性；冷却 20 年（7300 天），由 custom_trigger_tooltip 显示冷却原因

country_decisions = {

	acg_mec_e_rechoose_seven_archon = {
		potential = {
			religion = acg_mec_e_seven_archons
		}
		allow = {
			NOT = { has_regency = yes }
			custom_trigger_tooltip = {
				tooltip = acg_mec_e_rechoose_deity_cooldown_tooltip
				OR = {
					NOT = { has_country_flag = acg_mec_e_rechoose_deity_flag }
					had_country_flag = {
						flag = acg_mec_e_rechoose_deity_flag
						days = 7300
					}
				}
			}
		}
		effect = {
			country_event = { id = acg_mec_e_teyvat_religion.1 }
		}
		ai_will_do = {
			factor = 0
		}
	}

	acg_mec_e_rechoose_moon = {
		potential = {
			religion = acg_mec_e_moon_goddess
		}
		allow = {
			NOT = { has_regency = yes }
			custom_trigger_tooltip = {
				tooltip = acg_mec_e_rechoose_deity_cooldown_tooltip
				OR = {
					NOT = { has_country_flag = acg_mec_e_rechoose_deity_flag }
					had_country_flag = {
						flag = acg_mec_e_rechoose_deity_flag
						days = 7300
					}
				}
			}
		}
		effect = {
			country_event = { id = acg_mec_e_teyvat_religion.2 }
		}
		ai_will_do = {
			factor = 0
		}
	}
}
